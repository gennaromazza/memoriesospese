--- prima
import React from 'react'
// ...
// ❌ Vecchio approccio via fetch che causa CORS
async function fetchGalleryMetadata(code: string) {
  const resp = await fetch('https://us-central1-wedding-gallery-397b6.cloudfunctions.net/getGalleryMetadata', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ code }) // <- chiave sbagliata + chiamata REST
  })
  if (!resp.ok) throw new Error('failed')
  return resp.json()
}

--- dopo
import React from 'react'
import { getFunctions, httpsCallable } from 'firebase/functions'
import { app } from '../lib/firebase' // <-- percorso corretto alla tua init

async function fetchGalleryMetadata(code: string) {
  const fn = httpsCallable(
    getFunctions(app, 'us-central1'),
    'getGalleryMetadata'
  )
  const galleryCode = String(code).trim().toUpperCase()
  const { data } = await fn({ galleryCode })
  return data as {
    id: string
    name?: string
    code: string
    requiresSecurityQuestion: boolean
    securityQuestion?: string
  }
}
