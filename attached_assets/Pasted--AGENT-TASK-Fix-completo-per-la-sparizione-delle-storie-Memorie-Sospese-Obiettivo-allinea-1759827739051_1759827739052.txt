--- NEW FILE: src/hooks/useCoupleStory.ts ---
import { useState, useEffect } from "react";
import { doc, onSnapshot } from "firebase/firestore";
import { db } from "@/lib/firebase";
import type { CoupleStory } from "@shared/schema";

export function useCoupleStory(galleryId: string) {
  const [story, setStory] = useState<CoupleStory | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!galleryId) return;
    const ref = doc(db, "coupleStories", galleryId);
    const unsub = onSnapshot(ref, (snap) => {
      setStory(snap.exists() ? ({ id: snap.id, ...(snap.data() as any) }) : null);
      setLoading(false);
    }, () => setLoading(false));
    return () => unsub();
  }, [galleryId]);

  return { story, loading };
}
