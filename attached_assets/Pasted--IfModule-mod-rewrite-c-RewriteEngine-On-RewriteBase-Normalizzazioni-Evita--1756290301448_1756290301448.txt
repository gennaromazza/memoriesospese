<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # --- Normalizzazioni ---
  # Evita // doppi nel path
  RewriteCond %{REQUEST_URI} ^(.*/)//+(.*)$
  RewriteRule . %1/%2 [R=301,L]

  # --- Redirect specifici ---
  # /carnival2025 -> /carnival2025/
  RewriteCond %{REQUEST_URI} ^/carnival2025$ [NC]
  RewriteRule ^carnival2025$ /carnival2025/ [R=301,L]

  # Correggi typo /memomoriesospese -> /memoriesospese
  RewriteCond %{REQUEST_URI} ^/memomoriesospese(/.*)?$ [NC]
  RewriteRule ^memomoriesospese(/.*)?$ /memoriesospese$1 [R=301,L]

  # --- SPA fallback (routing client-side) ---
  # /wedgallery -> /wedgallery/index.html
  RewriteCond %{REQUEST_URI} ^/wedgallery(/.*)?$ [NC]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^wedgallery(/.*)?$ /wedgallery/index.html [L]

  # /carnival2025 -> /carnival2025/index.html
  RewriteCond %{REQUEST_URI} ^/carnival2025(/.*)?$ [NC]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^carnival2025(/.*)?$ /carnival2025/index.html [L]

  # /memoriesospese -> /memoriesospese/index.html   <<< NUOVO
  RewriteCond %{REQUEST_URI} ^/memoriesospese(/.*)?$ [NC]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^memoriesospese(/.*)?$ /memoriesospese/index.html [L]

  # --- Gestione asset per /wedgallery (se servono) ---
  RewriteCond %{REQUEST_URI} ^/assets/.*\.(css|js|png|jpg|jpeg|gif|ico|svg|woff2?|ttf|eot)$ [NC]
  RewriteCond %{HTTP_REFERER} ^https?://[^/]+/wedgallery(/|$) [NC]
  RewriteRule ^assets/(.*)$ /wedgallery/assets/$1 [L]

  RewriteCond %{REQUEST_URI} ^/(.*\.(css|js|png|jpg|jpeg|gif|ico|svg|woff2?|ttf|eot))$ [NC]
  RewriteCond %{HTTP_REFERER} ^https?://[^/]+/wedgallery(/|$) [NC]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteRule ^(.*)$ /wedgallery/$1 [L]

  # --- WordPress (sempre per ultime) ---
  RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
  RewriteRule ^index\.php$ - [L]
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule . /index.php [L]
</IfModule>
