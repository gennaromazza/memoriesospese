rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ===== Helper Functions =====
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function isAdmin() {
      return isSignedIn() && (
        request.auth.token.email == 'gennaro.mazzacane@gmail.com' ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
      );
    }

    function isValidRole(role) {
      return role == 'bride' || role == 'groom';
    }

    // ===== Users =====
    match /users/{userId} {
      allow read: if isSignedIn();
      allow write: if isOwner(userId) || isAdmin();
    }

    // ===== Subscriptions =====
    match /subscriptions/{userId} {
      allow read: if isOwner(userId) || isAdmin();
      allow write: if false; // Solo via webhook
    }

    // ===== Galleries =====
    match /galleries/{galleryId} {
      allow read: if true;
      allow create: if isSignedIn() || isAdmin();
      allow update: if isSignedIn() &&
        (resource.data.userId == request.auth.uid || isAdmin());
      allow delete: if isSignedIn() &&
        (resource.data.userId == request.auth.uid || isAdmin());

      // ----- Subcollections -----
      match /photos/{photoId} {
        allow read: if true;
        allow create: if isSignedIn();
        allow update, delete: if isSignedIn() &&
          (resource.data.uploaderEmail == request.auth.token.email ||
           resource.data.uploaderUid == request.auth.uid || isAdmin());
      }
// ====== COUPLE STORIES SYSTEM RULES ======

// Couple Stories collection - Admin only creation/modification, public read access for gallery viewing
match /coupleStories/{galleryId} {
  allow read: if true; // Public read access for story viewing in galleries
  allow create: if isAdmin(); // Only admin can create new stories
  allow update: if isAdmin(); // Only admin can update existing stories
  allow delete: if isAdmin(); // Only admin can delete stories
}
      match /comments/{commentId} {
        allow read: if true;
        allow create: if isSignedIn();
        allow update, delete: if isSignedIn() &&
          (resource.data.userId == request.auth.uid || isAdmin());
      }

      match /voiceMemos/{memoId} {
        allow read: if true;
        allow create: if true;
        allow update, delete: if isSignedIn() &&
          (resource.data.userId == request.auth.uid || isAdmin());
      }

      // ===== 🎯 QUESTIONNAIRES - ACCESSO COMPLETO OSPITI =====
      match /questionnaires/{questionnaireId} {
        // ✅ Lettura pubblica sempre permessa
        allow read: if true;
        // ✅ Creazione solo da admin autenticati  
        allow create: if isAdmin();
        // ✅ Aggiornamento permesso per ospiti (solo campi status e updatedAt)
        allow update: if isAdmin() || true; // Permetti aggiornamento pubblico
        allow delete: if isAdmin();

        // ===== 🎯 RISPOSTE FINALI - ACCESSO OSPITI =====
        match /answers/{role} {
          allow read: if true;
          allow write: if isValidRole(role); // Ospiti possono scrivere le proprie risposte
        }

        // ===== BOZZE RISPOSTE - ACCESSO OSPITI =====
        match /drafts/{role} {
          allow read: if true;
          allow write: if isValidRole(role); // Ospiti possono scrivere le proprie bozze
        }
      }
    }

    // ===== Global Photos =====
    match /photos/{photoId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow update: if isSignedIn() &&
        (resource.data.uploaderEmail == request.auth.token.email || isAdmin());
      allow delete: if isSignedIn() &&
        (resource.data.uploaderEmail == request.auth.token.email || isAdmin());
    }

    // ===== Comments =====
    match /comments/{commentId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow update: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow delete: if isSignedIn() &&
        (resource.data.userId == request.auth.uid || isAdmin());
    }

    // ===== Voice Memos =====
    match /voiceMemos/{memoId} {
      allow read: if true;
      allow create: if true;
      allow update: if isSignedIn() && resource.data.userId == request.auth.uid;
      allow delete: if isSignedIn() &&
        (resource.data.userId == request.auth.uid || isAdmin());
    }

    // ===== Likes =====
    match /likes/{likeId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow delete: if isSignedIn() && resource.data.userId == request.auth.uid;
    }

    // ===== Email Notifications =====
    match /email-notifications/{notificationId} {
      allow read: if isAdmin();
      allow create: if isSignedIn() &&
        request.resource.data.keys().hasAll(['email', 'galleryId']) &&
        request.resource.data.email is string &&
        request.resource.data.galleryId is string;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // ===== Gallery Access =====
    match /gallery-access/{accessId} {
      allow read: if isAdmin();
      allow create: if true;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // ===== Studio Settings =====
    match /studio-settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /settings/{documentId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ===== FAQ =====
    match /faqSets/{faqSetId} {
      allow read: if true;
      allow write: if isAdmin();
      allow create: if isAdmin() ||
        (request.resource.data.createdBy == 'system' &&
         request.resource.data.version == 1);
    }

    // ===== Questionnaire Tokens =====
    match /questionnaireTokens/{tokenId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ===== Validation Sessions =====
    match /validationSessions/{sessionId} {
      allow read: if true;
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    // ===== Distributed Locks =====
    match /distributedLocks/{lockId} {
      allow read: if true;
      allow create, update: if true;
      allow delete: if true;
    }

    // ===== Rate Limits =====
    match /rateLimits/{identifier} {
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }
  }
}
